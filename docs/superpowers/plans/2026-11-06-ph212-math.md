## The MathML reader (scratchpad, every content task)

Saved as `<scratchpad>/read_mathml.py`; run as `python -X utf8
read_mathml.py PAGE.html`. It flags, per `<math>`: `mathvariant` starting
`bold`; a character U+1D400 to U+1D7FF; an `<mrow>` of two or more children
whose last is an `<mo>` other than a fence or U+2061; an empty `<mrow>` as
a script; an `<mo>*</mo>`. It lists every `<mtext>` word. Exit 1 when
anything is flagged. It cannot see a word set as letters (`\mathrm{total}`
is one `<mi>` per letter, not flagged): the expected `mtext` list in each
task covers that.

```python
import re, sys
import xml.etree.ElementTree as ET

M = "{http://www.w3.org/1998/Math/MathML}"
# Fences, and U+2061 (function application, which Pandoc writes after \sin).
FENCES = set("()[]{}|‖⟨⟩⁡")
SCRIPTS = {"msub", "msup", "msubsup", "munder", "mover", "munderover"}

def local(el):
    return el.tag.split("}")[-1]

def flags(math):
    out = []
    for el in math.iter():
        tag = local(el)
        if tag in ("semantics", "annotation"):
            continue
        if el.get("mathvariant", "").startswith("bold"):
            out.append(f"mathvariant={el.get('mathvariant')} on {tag} {el.text!r}")
        if el.text and any(0x1D400 <= ord(c) <= 0x1D7FF for c in el.text):
            out.append(f"math-bold character in {tag} {el.text!r}")
        kids = [k for k in el if local(k) != "annotation"]
        if tag == "mrow" and len(kids) > 1:
            last = kids[-1]
            if local(last) == "mo" and (last.text or "").strip() not in FENCES:
                out.append(f"mrow ends on operator {last.text!r}")
        if tag == "mo" and (el.text or "").strip() == "*":
            out.append("typed * read as star")
        if tag in SCRIPTS:
            for k in kids[1:]:
                if local(k) == "mrow" and len(k) == 0:
                    out.append(f"empty script in {tag}")
    return out

def main(path):
    html = open(path, encoding="utf-8").read()
    maths = re.findall(r"<math\b.*?</math>", html, re.S)
    flagged, words = 0, []
    for i, src in enumerate(maths, 1):
        src = src.replace("&nbsp;", " ")
        math = ET.fromstring(src if "xmlns" in src[:80] else
                             src.replace("<math", f'<math xmlns="{M[1:-1]}"', 1))
        ann = math.find(f".//{M}annotation")
        tex = ann.text if ann is not None else "?"
        words += [el.text for el in math.iter(M + "mtext") if el.text]
        f = flags(math)
        if f:
            flagged += 1
            print(f"{i:3} {tex!r}: " + "; ".join(f))
    print(f"{len(maths)} equations, {flagged} flagged; mtext words: {words}")
    return 1 if flagged else 0

if __name__ == "__main__":
    sys.exit(main(sys.argv[1]))
```

```python
def apply_ruled_equations(o_eqs, c_eqs, ruled):
    """Excuse each declared equation's text and structure (D-0167).

    A declaration names an equation by Word's text and the converted text,
    as compare_equations prints them; it excuses the first position not yet
    excused where both match, by giving the converted record the original's
    text and structure. Its kind and the count are still compared.
    """
    problems, excused = [], set()
    for item in ruled or []:
        if not item.get("ledger"):
            problems.append(f"ruled_equations entry lacks a ledger reference:"
                            f" {item.get('original')!r}")
        want = (math_norm(item.get("original") or ""), math_norm(item.get("converted") or ""))
        at = next((i for i, (o, c) in enumerate(zip(o_eqs, c_eqs))
                   if i not in excused and (o["norm"], c["norm"]) == want), None)
        if at is None:
            problems.append(f"ruled_equations entry matches no equation:"
                            f" {item.get('original')!r} -> {item.get('converted')!r}")
            continue
        excused.add(at)
        o, c = o_eqs[at], c_eqs[at]
        if o["norm"] == c["norm"] and o["shape"] in (None, c["shape"]):
            problems.append(f"ruled_equations entry for equation {at + 1} is needless:"
                            f" it passes undeclared ({c['tex']!r})")
        c_eqs[at] = {**c, "text": o["text"], "norm": o["norm"], "shape": o["shape"]}
    return problems
```
